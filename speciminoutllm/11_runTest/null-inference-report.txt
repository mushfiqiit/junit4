// A) Annotation decisions
//
// | location | inferred annotation | one-sentence reason |
// | --- | --- | --- |
// | local variable `Method runMethod` in `TestCase.runTest` | `@Nonnull` | The path through the `NoSuchMethodException` handler is intended to terminate via `fail`, so the dereference of `runMethod` is reached only with a non-null value. |
//
// B) Cause of the warning
//
// NullAway warns at the dereference `runMethod.getModifiers()` because `runMethod` is initialized to `null` and NullAway does not infer that the `fail(...)` call in the `NoSuchMethodException` handler always terminates.
// Therefore, at the join point after the `try`/`catch`, NullAway conservatively considers the previously assigned `null` still live, making the dereference unsafe.
// Annotating the local variable as `@Nonnull` marks the variable as non-null for the dereference check; any separate warning about the `null` initializer is out of scope for this one warning.

package junit.framework;

import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;
import java.lang.reflect.Modifier;
import javax.annotation.Nonnull;

@SuppressWarnings("deprecation")
public abstract class TestCase extends Assert implements Test {

  private String fName;

  protected void runTest() throws Throwable {
    assertNotNull("TestCase.fName cannot be null", fName);
    @Nonnull Method runMethod = null;
    try {
      runMethod = getClass().getMethod(fName, (Class[]) null);
    } catch (NoSuchMethodException e) {
      fail("Method \"" + fName + "\" not found");
    }
    if (!Modifier.isPublic(runMethod.getModifiers())) {
      fail("Method \"" + fName + "\" should be public");
    }
    try {
      runMethod.invoke(this);
    } catch (InvocationTargetException e) {
      e.fillInStackTrace();
      throw e.getTargetException();
    } catch (IllegalAccessException e) {
      e.fillInStackTrace();
      throw e;
    }
  }

  public static void fail(String message) {
    throw new java.lang.Error();
  }

  public static void assertNotNull(String message, Object object) {
    throw new java.lang.Error();
  }
}

// A) Annotation decisions
//
// location                                                                 inferred annotation   one-sentence reason
// -----------------------------------------------------------------------   -------------------   -------------------------------------------------------------
// org.junit.internal.runners.ErrorReportingRunner#getCauses(Throwable)      @Nullable             The recursive call may pass `cause.getCause()`, and `InvocationTargetException.getCause()` may be null.
//
// No other location is annotated; unannotated types remain @Nonnull by default.
//
// B) Cause of the warning
//
// The warning is on the recursive call `getCauses(cause.getCause())`.
// `cause` is non-null in the `instanceof InvocationTargetException` branch, so calling
// `cause.getCause()` is safe, but the returned `Throwable` is @Nullable according to the
// JDK annotation. The recursive parameter `cause` in `getCauses(Throwable cause)` is
// unannotated, hence @NonNull, so passing the @Nullable cause is unsafe. Annotating the
// parameter as @Nullable makes the parameter accept a null cause and resolves the warning.
//
// === org/junit/internal/runners/ErrorReportingRunner.java ===
package org.junit.internal.runners;

import static java.util.Collections.singletonList;

import java.lang.reflect.InvocationTargetException;
import java.util.List;
import javax.annotation.Nullable;
import org.junit.runner.Runner;
import org.junit.runners.model.InitializationError;
import org.junit.runners.model.InvalidTestClassError;

public class ErrorReportingRunner extends Runner {

  @SuppressWarnings("deprecation")
  private List<Throwable> getCauses(@Nullable Throwable cause) {
    if (cause instanceof InvocationTargetException) {
      return getCauses(cause.getCause());
    }
    if (cause instanceof InvalidTestClassError) {
      return singletonList(cause);
    }
    if (cause instanceof InitializationError) {
      return ((InitializationError) cause).getCauses();
    }
    if (cause instanceof org.junit.internal.runners.InitializationError) {
      return ((org.junit.internal.runners.InitializationError) cause).getCauses();
    }
    return singletonList(cause);
  }
}
